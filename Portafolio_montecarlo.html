# Creación de una función para extrar la información de precios de cierre de las acciones
def obtener_precios_cierre(tickers, fecha_inicio, fecha_fin):
    data = pd.DataFrame()
    for ticker in tickers:
        precios_cierre = yf.download(ticker, start = fecha_inicio, end = fecha_fin)['Adj Close']
        data[ticker] = precios_cierre
    return data

Simulación Montecarlo¶

La simulación de Montecarlo en la optimización de portafolios sirve para explorar y analizar posibles combinaciones de activos de manera aleatoria y evaluar su desempeño en términos de retorno y riesgo.

In [86]:
# Parámetros
nro_portafolios = 5000    # Cantidad de portafolios a generar
tasa_libre_riesgo = 0.05  # Tasa libre de riesgo (Supongamos que sea del 5%)

# Arrays para almacenar resultados: Imicialmente, tendrán ceros y cada vez que se genere un nuevo portafolio van a cambiar continuamente
pesos_global = np.zeros((nro_portafolios, len(tickers))) # Almacenará pesos
array_retornos = np.zeros(nro_portafolios) # Almacenará los retornos
array_riesgo = np.zeros(nro_portafolios) # Almacenará los riesgos
array_sharpe = np.zeros(nro_portafolios) # Almacenará los ratios Sharpe

# Simulación
for port in range(nro_portafolios):
    # Generando pesos aleatorios
    pesos = np.random.random(len(tickers))
    pesos /= np.sum(pesos)
    pesos_global[port, :] = pesos

    # Cálculo de retorno esperado del portafolio
    retorno_portafolio = np.sum(retornos.mean() * pesos) * 252 # Se elige 252 porque son el total de días que el mercado de valores está "abierto"
    array_retornos[port] = retorno_portafolio

    # Cálculo de riesgo del portafolio
    riesgo_portafolio = np.sqrt(np.dot(pesos.T, np.dot(retornos.cov() * 252, pesos)))
    array_riesgo[port] = riesgo_portafolio

    # Cálculo de Ratios Sharpe
    array_sharpe[port] = (retorno_portafolio - tasa_libre_riesgo) / riesgo_portafolio
